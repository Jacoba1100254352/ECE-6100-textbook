\chapter{Formula Reference}\label{ch:formula-compendium}
This appendix gathers the core formulas used across Chapters 1--10.
Keep the stated model with each equation: symbols are reused across topics, latency units must agree, and overlapping costs must not be added twice.


\section*{Chapter 1 Formula Set}\label{sec:auto-056}

\paragraph{Chapter 1-F1 (Vector Serial vs Pipelined Cycles)}\label{par:auto-043}
\begin{equation}\label{eq:auto-054}
    \text{serial cycles}=nk,\qquad
    \text{pipeline cycles}\approx k+n-1.
\end{equation}
Here $n$ is the number of elements and $k$ is the per-element latency in cycles; the pipeline accepts one element per cycle without stalls.


\section*{Chapter 2 Formula Set}\label{sec:auto-057}

\paragraph{Chapter 2-F1 (CPU Time Decomposition)}\label{par:auto-044}
\begin{equation}\label{eq:auto-055}
    \text{CPU time}=\text{IC}\times\text{CPI}\times\text{CT}.
\end{equation}

\paragraph{Chapter 2-F2 (Frequency Form)}\label{par:auto-045}
\begin{equation}\label{eq:auto-056}
    \text{CPU time}=\frac{\text{IC}\times\text{CPI}}{f}.
\end{equation}
Here $f$ is clock frequency; it is not the accelerated time fraction used in Amdahl's law below.

\paragraph{Chapter 2-F3 (Instruction-Mix Total Cycles)}\label{par:auto-046}
\begin{equation}\label{eq:auto-057}
    \text{Total cycles}=\sum_i\left(\text{IC}_i\times\text{CPI}_i\right).
\end{equation}

\paragraph{Chapter 2-F4 (Average CPI)}\label{par:auto-047}
\begin{equation}\label{eq:auto-058}
    \text{CPI}_{avg}=\frac{\text{Total cycles}}{\text{Total instructions}}.
\end{equation}

\paragraph{Chapter 2-F5 (Speedup Definition)}\label{par:auto-048}
\begin{equation}\label{eq:auto-059}
    S=\frac{T_{old}}{T_{new}}.
\end{equation}

\paragraph{Chapter 2-F6 (Amdahl's Law)}\label{par:auto-049}
\begin{equation}\label{eq:auto-060}
    S_{Amdahl}=\frac{1}{(1-f)+\frac{f}{s}}.
\end{equation}
$f$ is the fraction of \emph{baseline execution time} accelerated by factor $s$; the fixed-work model ignores new overhead and overlap changes.

\paragraph{Chapter 2-F7 (Amdahl Upper Bound)}\label{par:auto-050}
\begin{equation}\label{eq:auto-061}
    S_{\text{upper}}=\frac{1}{1-f}.
\end{equation}
For $f<1$ as $s\rightarrow\infty$.

\paragraph{Chapter 2-F8 (Gustafson's Law)}\label{par:auto-051}
\begin{equation}\label{eq:auto-062}
    S_{Gustafson}=\alpha+(1-\alpha)N.
\end{equation}
$\alpha$ is the serial fraction measured in the scaled $N$-processor run, not the baseline fraction from Amdahl's law.

\paragraph{Chapter 2-F9 (Equal-Weight Mean of Times)}\label{par:auto-052}
\begin{equation}\label{eq:auto-063}
    \bar{T}_{arith}=\frac{1}{n}\sum_{i=1}^{n}T_i.
\end{equation}

\paragraph{Chapter 2-F10 (Combined Rate for Equal-Work Segments)}\label{par:auto-053}
\begin{equation}\label{eq:auto-064}
    \bar{R}_{harm}=\frac{n}{\sum_{i=1}^{n}\frac{1}{R_i}}.
\end{equation}
For positive rates on consecutive equal-work segments. Equal-time segments instead use an arithmetic rate mean; in general, divide total work by total time.


\section*{Chapter 3 Formula Set}\label{sec:auto-058}

\paragraph{Chapter 3-F1 ($(C,B,S)$ Definitions)}\label{par:auto-054}
\begin{equation}\label{eq:auto-065}
    \text{capacity}=2^C\ \text{bytes},\qquad
    \text{block size}=2^B\ \text{bytes},\qquad
    \text{associativity}=2^S\ \text{ways}.
\end{equation}
Capacity here means data bytes, excluding tags and control metadata. The simple byte-addressed geometry requires $C\ge B+S$.

\paragraph{Chapter 3-F2 (Total Blocks and Sets)}\label{par:auto-055}
\begin{equation}\label{eq:auto-066}
    \text{blocks}=2^{C-B},\qquad
    \text{sets}=\frac{2^C}{2^B\cdot 2^S}=2^{C-B-S}.
\end{equation}
\begin{equation}\label{eq:auto-087}
    \text{lines}=\text{sets}\cdot\text{ways}=2^{C-B-S}\cdot 2^S=2^{C-B}.
\end{equation}
Direct-mapped corollary: if $S=0$, then $\text{ways}=1$ and $\text{lines}=\text{sets}$.
Set-associative corollary: if $S>0$, then $\text{lines}>\text{sets}$.

\paragraph{Chapter 3-F3 (Address-Field Widths, General $AW$)}\label{par:auto-056}
\begin{equation}\label{eq:auto-067}
    |\text{Offset}|=B,\qquad
    |\text{Index}|=C-B-S,\qquad
    |\text{Tag}|=AW-(C-S).
\end{equation}
\begin{equation}\label{eq:auto-068}
    \text{Address}=[\text{Tag}\mid\text{Index}\mid\text{Offset}].
\end{equation}
$AW$ is the address width used by the modeled lookup, not automatically the ISA register width or the logarithm of installed RAM size.

\paragraph{Chapter 3-F4 (Address-Field Widths, 32-bit Address)}\label{par:auto-057}
\begin{equation}\label{eq:auto-069}
    |\text{Offset}|=B,\qquad
    |\text{Index}|=C-B-S,\qquad
    |\text{Tag}|=32-(C-S)=32-C+S.
\end{equation}

\paragraph{Chapter 3-F5 (Way Selection Relationship)}\label{par:auto-058}
\begin{equation}\label{eq:auto-070}
    \text{ways}=2^S,\qquad
    \log_2(\text{ways})=S.
\end{equation}

\paragraph{Chapter 3-F6 (Single-Level Average Access Time)}\label{par:auto-059}
\begin{equation}\label{eq:auto-071}
    AAT=HT+MR\cdot MP.
\end{equation}
$MP$ is the additional service time after the current level's hit lookup in this additive model.

\paragraph{Chapter 3-F7 (Bit Overhead from Valid/Dirty)}\label{par:auto-060}
\begin{equation}\label{eq:auto-072}
    \text{overhead fraction}=\frac{n_v+n_d}{2^{B+3}},\qquad
    n_v=1,\quad n_d\in\{0,1\}.
\end{equation}
Valid/dirty bits per block divided by data bits per block; this does not include tag or replacement-state overhead.

\paragraph{Chapter 3-F8 (Full-Block Miss Penalty Model)}\label{par:auto-061}
\begin{equation}\label{eq:auto-073}
    MP_{full}=T_1+(N_{sb}-1)T_f,\qquad N_{sb}=\left\lceil 2^B/W\right\rceil.
\end{equation}
$W$ is bytes per transfer, $T_1$ is time to the first completed sub-block, and $T_f$ is the interval between later sub-blocks.

\paragraph{Chapter 3-F9 (Early-Restart Approximation)}\label{par:auto-062}
\begin{equation}\label{eq:auto-074}
    MP_{ER}\approx T_1+\frac{N_{sb}-1}{2}T_f.
\end{equation}
Assumes an in-order fill and a uniform distribution over requested sub-block positions.

\paragraph{Chapter 3-F10 (Recursive Two-Level Form)}\label{par:auto-063}
\begin{equation}\label{eq:auto-075}
    AAT_{L1}=HT_{L1}+MR_{L1}\cdot AAT_{L2}.
\end{equation}
The L2 term is complete service time per L1 miss; L2 miss ratios are conditional on accesses reaching L2 in this serial-lookup model.

\paragraph{Chapter 3-F11 (VIPT Constraint)}\label{par:auto-064}
\begin{equation}\label{eq:auto-076}
    C-S\le P.
\end{equation}
$2^P$ bytes per page; this is the conventional VIPT index-within-page-offset constraint without extra synonym-handling machinery.


\section*{Chapter 4 Formula Set}\label{sec:auto-059}
For the pipeline timing formulas, $N$ is the number of balanced stages, $M$ is the instruction count, $T_s$ is stage logic delay, and $T_{reg}$ is register overhead; assume one new instruction per cycle and no stalls unless stated.

\paragraph{Chapter 4-F1 (Pipelined Clock Approximation)}\label{par:auto-065}
\begin{equation}\label{eq:auto-077}
    T_{clk,pipe}\approx T_s+T_{reg}.
\end{equation}

\paragraph{Chapter 4-F2 (Ideal Long-Run Pipeline Speedup)}\label{par:auto-066}
\begin{equation}\label{eq:auto-078}
    S_{ideal}\approx\frac{N T_s}{T_s+T_{reg}}.
\end{equation}

\paragraph{Chapter 4-F3 (Finite-Length Pipelined Time)}\label{par:auto-067}
\begin{equation}\label{eq:auto-079}
    T_{pipe}\approx(N+M-1)T_{clk,pipe}.
\end{equation}

\paragraph{Chapter 4-F4 (Unpipelined Time for $M$ Instructions)}\label{par:auto-068}
\begin{equation}\label{eq:auto-080}
    T_{unpiped}\approx MNT_s.
\end{equation}

\paragraph{Chapter 4-F5 (Practical CPI Model)}\label{par:auto-069}
\begin{equation}\label{eq:auto-081}
    CPI=1+\text{stalls}_{avg}.
\end{equation}
For the single-issue model with ideal CPI 1, counting non-overlapping added stall cycles per retired instruction.

\paragraph{Chapter 4-F6 (Branch-Stall CPI Contribution)}\label{par:auto-070}
\begin{equation}\label{eq:auto-082}
    CPI\approx 1+f_{branch}\cdot \text{stall}_{branch,avg}.
\end{equation}

\paragraph{Chapter 4-F7 (History-Table Indexing)}
\begin{align}
    index_{GShare}&=PC_{selected}\oplus GHR,\label{eq:app-b-gshare}\\
    index_{GSelect}&=\operatorname{concat}(PC_p,GHR_h),\qquad N_{counters}=2^{p+h}.\label{eq:app-b-gselect}
\end{align}
For GShare, the selected PC and history fields have equal width.
Exclude alignment bits when the predictor's indexing convention requires it.

\paragraph{Chapter 4-F8 (Perceptron Prediction and Training)}
\begin{equation}\label{eq:app-b-perceptron}
    y=w_0+\sum_{i=1}^{n}w_i x_i,\qquad x_i\in\{-1,+1\}.
\end{equation}
Predict taken when $y\ge0$.
If wrong or $|y|\le\theta$, update $w_0\leftarrow w_0+t$ and $w_i\leftarrow w_i+t x_i$, where $t$ encodes the actual outcome.
The exercises use $\theta=\lfloor1.93n+14\rfloor$ and assume no weight overflow.


\section*{Chapter 5 and Simulation Formula Set}\label{sec:auto-060}

\paragraph{Chapter 5-F1 (Two-Level AAT with Explicit L1 Penalty)}\label{par:auto-071}
\begin{equation}\label{eq:auto-083}
    AAT_{L1}=HT_{L1}+MR_{L1}\cdot MP_{L1},\qquad
    MP_{L1}=AAT_{L2}.
\end{equation}

\paragraph{Chapter 5-F2 (L2 AAT Model)}\label{par:auto-072}
\begin{equation}\label{eq:auto-084}
    AAT_{L2}=HT_{L2}+MR_{L2,read}\cdot DRAM\_TIME.
\end{equation}

\paragraph{Chapter 5-F3 (DRAM Block Service Model)}\label{par:auto-073}
\begin{equation}\label{eq:auto-085}
    DRAM\_TIME=DRAM\_AT+DRAM\_AT\_PER\_WORD\cdot WORDS\_PER\_BLOCK.
\end{equation}
Here $DRAM\_AT$ excludes the per-word transfer time; Chapter 3's $T_1$ includes the first completed sub-block transfer.


\section*{Chapter 6 Formula Set (Dynamic ILP, Superscalar, and Out-of-Order Execution)}\label{sec:ch6-formulas}

\paragraph{Chapter 6-F1 (IPC and CPI Relationship)}\label{par:ch6-formula-1}
\begin{equation}\label{eq:app-ch6-formula-ipc-cpi}
    IPC=\frac{1}{CPI}.
\end{equation}
Use the same instruction count and cycle interval for both quantities, normally retired instructions for architectural IPC.

\paragraph{Chapter 6-F2 (CPI/IPC Threshold Equivalence)}\label{par:ch6-formula-2}
\begin{equation}\label{eq:app-ch6-formula-threshold}
    CPI<1 \iff IPC>1.
\end{equation}

\paragraph{Chapter 6-F3 (Retire-Throughput Bound)}\label{par:ch6-formula-3}
\begin{equation}\label{eq:app-ch6-formula-throughput}
    IPC_{retire}\le \min\!\left(IPC_{fetch},\ IPC_{dispatch},\ IPC_{execute},\ IPC_{state\ update}\right).
\end{equation}
A sustained-rate bound, not a claim that every short window has matching fetch and retirement counts.

\paragraph{Chapter 6-F4 (Simplified Scoreboard Fire Condition)}\label{par:ch6-formula-4}
\begin{equation}\label{eq:app-ch6-formula-fire}
    \text{Fire if } \neg busy(src_1)\wedge \neg busy(src_2)\wedge \neg busy(dest)\wedge \neg busy(FU_{needed}).
\end{equation}
The simple FICO model captures sources before younger writers pass. A busy FU means unable to accept another operation, not merely occupied by an older pipelined operation.

\paragraph{Chapter 6-F5 (Result-Bus Completion Bound)}\label{par:ch6-formula-5}
\begin{equation}\label{eq:app-ch6-formula-result-bus}
    IPC_{complete}\le N_{result\ buses}.
\end{equation}
Applies to result producers that each require one broadcast, not necessarily to stores or branches using other completion paths.

\paragraph{Chapter 6-F6 (Physical Register File Port Estimate)}\label{par:ch6-formula-6}
\begin{equation}\label{eq:app-ch6-formula-prf-ports}
    N_{data\ read\ ports}\approx 2W_{issue},\qquad
    N_{write\ ports}\approx N_{result\ buses}.
\end{equation}
For a simple full-bandwidth PRF with two source values per issued operation.
Polling two ready bits for every scheduling entry instead costs up to $2N_{sched}$ one-bit checks per cycle.

\paragraph{Chapter 6-F7 (Precise Restartable-Fault Property)}\label{par:ch6-formula-7}
\begin{equation}\label{eq:app-ch6-formula-precise}
    \begin{aligned}
        \text{restartable fault at }i \Rightarrow\ &\text{all older instructions are committed}\\
        &\wedge\ \text{$i$ and all younger instructions have no committed effects}
    \end{aligned}
\end{equation}

\paragraph{Chapter 6-F8 (RAT Rename Mapping)}\label{par:ch6-formula-8}
\begin{equation}\label{eq:app-ch6-formula-rat}
    P_{src1}=RAT[A_{src1}],\qquad
    P_{src2}=RAT[A_{src2}],\qquad
    RAT[A_{dest}]\leftarrow P_{free}.
\end{equation}

\paragraph{Chapter 6-F9 (Ready-Based Fire Condition)}\label{par:ch6-formula-9}
\begin{equation}\label{eq:app-ch6-formula-fire-ready}
    fire_i \Leftarrow ready(P_{src1})\wedge ready(P_{src2})\wedge free(FU_i).
\end{equation}

\paragraph{Chapter 6-F10 (ROB Retirement Condition)}\label{par:ch6-formula-10}
\begin{equation}\label{eq:app-ch6-formula-retire}
    retire \Leftarrow ready(ROB_{head})\wedge \neg exception(ROB_{head}).
\end{equation}

\paragraph{Chapter 6-F11 (ROB Exception Rule)}\label{par:ch6-formula-11}
\begin{equation}\label{eq:app-ch6-formula-exception}
    \begin{aligned}
        exception(ROB_{head}) \Rightarrow\ &\text{flush younger instructions}\\
        &\wedge\ \text{transfer control to the retry or handler PC}
    \end{aligned}
\end{equation}


\section*{Chapter 7 Formula Set (VLIW and Software Pipelining)}\label{sec:ch7-formulas}

\paragraph{Chapter 7-F1 (Dependence Height)}\label{par:ch7-formula-1}
\begin{equation}\label{eq:app-ch7-formula-height}
    height(i)=
    \begin{cases}
        lat(i), & succ(i)=\emptyset\\
        lat(i)+\max\limits_{j\in succ(i)} height(j), & \text{otherwise}
    \end{cases}
\end{equation}
For the acyclic graph and latency convention used in Chapter 7; compact cyclic loop graphs instead need recurrence analysis.

\paragraph{Chapter 7-F2 (Earliest Legal List-Scheduling Cycle)}\label{par:ch7-formula-2}
\begin{equation}\label{eq:app-ch7-formula-list-legal}
    cycle(i)=\min\!\left\{c\ \middle|\ \forall j\in pred(i),\ c\ge cycle(j)+\delta_{ji},\ \text{resources are legal in }c \right\}.
\end{equation}
$\delta_{ji}$ is the required edge separation; for a RAW edge in the simple fixed-latency model, it is $lat(j)$.

\paragraph{Chapter 7-F3 (Resource-Constrained Initiation Interval)}\label{par:ch7-formula-3}
\begin{equation}\label{eq:app-ch7-formula-res-ii}
    ResII=\max_t \left\lceil \frac{N_t}{U_t} \right\rceil
\end{equation}
Assumes disjoint classes of units that each accept one operation per cycle; multi-cycle resource occupancy requires a richer reservation model.

\paragraph{Chapter 7-F4 (Recurrence-Constrained Initiation Interval)}\label{par:ch7-formula-4}
\begin{equation}\label{eq:app-ch7-formula-rec-ii}
    RecII=\max_c \left\lceil \frac{Lat(c)}{Dist(c)} \right\rceil
\end{equation}
For dependence cycles with positive total iteration distance, using the sum of required edge latencies.

\paragraph{Chapter 7-F5 (Minimum Initiation Interval Bound)}\label{par:ch7-formula-5}
\begin{equation}\label{eq:app-ch7-formula-min-ii}
    MII=\max(ResII,RecII),\qquad II\ge MII.
\end{equation}
The lower bound need not be achievable by a legal schedule or by a bounded heuristic search.


\section*{Chapter 8 Formula Set (Multiprocessors and Coherence)}\label{sec:ch8-formulas}

\paragraph{Chapter 8-F1 (Ideal Embarrassingly Parallel Scaling)}\label{par:ch8-formula-1}
\begin{equation}\label{eq:ch8-formula-ideal-scaling}
    S(N)=N \qquad \text{for ideally balanced, independent work with no added overhead.}
\end{equation}
This idealization excludes communication, synchronization, serial work, and capacity or bandwidth bottlenecks.

\paragraph{Chapter 8-F2 (Directory Entry Count)}\label{par:ch8-formula-2}
\begin{equation}\label{eq:ch8-formula-dir-entries}
    N_{entries}=\frac{2^m}{2^v}=2^{m-v},
\end{equation}
For a dense directory covering all $2^m$ bytes of memory with $2^v$-byte blocks.

\paragraph{Chapter 8-F3 (Full Bit-Vector Directory Storage Estimate)}\label{par:ch8-formula-3}
\begin{equation}\label{eq:ch8-formula-full-dir}
    bits_{dir}\approx 2^{m-v}\left(N+n_{ctrl}\right),
\end{equation}
where $N$ is the number of PEs and $n_{ctrl}$ represents the small set of non-sharer control bits such as dirty, valid, or overflow indicators.

\paragraph{Chapter 8-F4 (Limited $K$-Pointer Directory Storage Estimate)}\label{par:ch8-formula-4}
\begin{equation}\label{eq:ch8-formula-kdir}
    bits_{dir}\approx 2^{m-v}\left(K\left\lceil \log_2 N \right\rceil + n_{ctrl}\right).
\end{equation}


\section*{Chapter 9 Formula Set (Synchronization and Memory Consistency)}\label{sec:ch9-formulas}

\paragraph{Chapter 9-F1 (Fetch-and-Op Semantics)}\label{par:ch9-formula-1}
\begin{equation}\label{eq:app-ch9-formula-fetchop}
    \text{\texttt{fetch\_and\_op}}(x,c):\qquad old=x,\qquad x\leftarrow x\;\textit{op}\;c,\qquad \text{return } old.
\end{equation}

\paragraph{Chapter 9-F2 (Dynamic Work Claim via Fetch-and-Add)}\label{par:ch9-formula-2}
\begin{equation}\label{eq:app-ch9-formula-work-claim}
    i=\text{\texttt{fetch\_and\_add}}(next,1),\qquad \text{execute iteration } i \text{ iff } i<N.
\end{equation}

\paragraph{Chapter 9-F3 (Last Arrival in a Counter-Based Barrier)}\label{par:ch9-formula-3}
\begin{equation}\label{eq:app-ch9-formula-barrier}
    \text{last arriver if }\text{\texttt{fetch\_and\_add}}(count,1)=N-1.
\end{equation}
This identifies the last arrival; a reusable barrier also requires phase tracking and memory-ordering semantics, as in Chapter~\ref{ch:sync-consistency}.


\section*{Chapter 10 Formula Set (Parallel Interconnection Networks)}\label{sec:ch10-formulas}

\paragraph{Chapter 10-F1 (K-ary N-Dimensional Node Count)}\label{par:ch10-formula-1}
\begin{equation}\label{eq:app-ch10-formula-kary}
    N_{nodes}=k^n.
\end{equation}

\paragraph{Chapter 10-F2 (Hypercube Size and Degree)}\label{par:ch10-formula-2}
\begin{equation}\label{eq:app-ch10-formula-hypercube}
    N_{nodes}=2^d,\qquad degree=d.
\end{equation}

\paragraph{Chapter 10-F3 (Bisection Bandwidth for Uniform Links)}\label{par:ch10-formula-3}
\begin{equation}\label{eq:app-ch10-formula-bisection}
    B_{bisect}=N_{cut}\times W_{link}.
\end{equation}
Uses the minimum equal-halves cut. Chapter 10 reports one-way capacity with $W_{link}$ specified per direction of each full-duplex edge.

\paragraph{Chapter 10-F4 (Omega Network Stage Count)}\label{par:ch10-formula-4}
\begin{equation}\label{eq:app-ch10-formula-omega}
    stages=\log_2 N.
\end{equation}
For $N$ a power of two and 2-by-2 switching elements.

\paragraph{Chapter 10-F5 (Whole-Packet Latency in the Unit-Link Model)}\label{par:ch10-formula-5}
\begin{equation}\label{eq:app-network-packet-latency}
    T_{\mathrm{SF}}=HF,\qquad T_{\mathrm{CT}}=H+F-1.
\end{equation}
$H$ one-cycle links, one flit per link per cycle, $F$ flits per packet, no extra router delay, and no contention; measure from first transmission start to final-flit arrival.
SF denotes store-and-forward; CT denotes cut-through.
